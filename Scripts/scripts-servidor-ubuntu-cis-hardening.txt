6. PRUEBA DE CONECTIVIDAD SSH A LOS EQUIPOS
-------------------------------------------
# SSH al FortiGate (probado durante la practica):
ssh admin@10.21.40.129

# Conexion TCP basica al puerto 22, sin autenticacion:
python3 - <<'PY'
import socket
for nombre, ip in [
    ('Router R-CIS', '10.21.40.1'),
    ('Switch SW-CIS', '10.21.40.130'),
    ('FortiGate', '10.21.40.129'),
]:
    try:
        with socket.create_connection((ip, 22), timeout=30):
            print(f'{nombre}: TCP/22 accesible')
    except OSError as error:
        print(f'{nombre}: sin respuesta TCP/22 ({error})')
PY

# NOTA: el Cisco IOS 12.4 utiliza algoritmos SSH antiguos; para las
# pruebas de automatizacion se utilizo Paramiko 3.5.1 en este entorno.
